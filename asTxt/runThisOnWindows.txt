' ─────────────────────────────────────────────
'  MCU Uploader IDE by Naph — Auto-Bootstrap Launcher
'  Downloads Python if needed, installs deps,
'  and launches the GUI — fully unattended.
' ─────────────────────────────────────────────

Dim fso, shell, scriptDir
Set fso   = CreateObject("Scripting.FileSystemObject")
Set shell = CreateObject("WScript.Shell")

' ── Locate this file's own directory ──
scriptDir = fso.GetParentFolderName(WScript.ScriptFullName) & "\"

' ── Ensure Arduino IDE is installed ──
If Not IsArduinoIdeInstalled() Then
    Dim ans
    ans = MsgBox("Arduino IDE is not installed on this computer, or its location was not detected." & vbCrLf & vbCrLf & _
                 "• Click YES to manually locate 'Arduino IDE.exe' or 'arduino.exe'." & vbCrLf & _
                 "• Click NO to install Arduino IDE using the offline installer MSI." & vbCrLf & _
                 "• Click CANCEL to exit the application.", _
                 vbYesNoCancel + vbQuestion, "Arduino IDE Not Found")
                 
    If ans = vbYes Then
        Dim selectedPath
        selectedPath = ChooseArduinoExe()
        If selectedPath <> "" Then
            ' Save the path to arduino_ide_path.txt
            Dim txtStream
            Set txtStream = fso.CreateTextFile(scriptDir & "arduino_ide_path.txt", True)
            txtStream.WriteLine selectedPath
            txtStream.Close
            MsgBox "Arduino IDE path saved successfully: " & selectedPath, vbInformation, "Success"
        Else
            MsgBox "No file selected. Exiting...", vbExclamation, "Error"
            WScript.Quit 1
        End If
    ElseIf ans = vbNo Then
        InstallArduinoMsi(scriptDir)
        ' Check again after installation
        If Not IsArduinoIdeInstalled() Then
            MsgBox "Arduino IDE still could not be detected after installation. Exiting...", vbCritical, "Error"
            WScript.Quit 1
        End If
    Else
        WScript.Quit 1
    End If
End If

' ── Find the bootstrap script ──
Dim bootstrapFile
bootstrapFile = scriptDir & "bootstrap.py"
If Not fso.FileExists(bootstrapFile) Then
    MsgBox "bootstrap.py not found in:" & vbCrLf & scriptDir & vbCrLf & vbCrLf & _
           "Please make sure bootstrap.py is in the same folder as this launcher.", _
           vbCritical, "MCU Uploader IDE by Naph"
    WScript.Quit 1
End If


' ═════════════════════════════════════════════
'  FIND SYSTEM PYTHON (to run bootstrap if needed)
' ═════════════════════════════════════════════
Dim systemPython, result
systemPython = ""

' Check for our own portable python first as a system fallback
Dim portablePython
portablePython = scriptDir & "_python\python.exe"
If fso.FileExists(portablePython) Then
    systemPython = portablePython
End If

' 1. Dynamically scan known Python install directories
'    (no hardcoded version numbers — works for any future Python)
If systemPython = "" Then
    Dim localApp, progFiles, progFiles86
    localApp   = shell.ExpandEnvironmentStrings("%LOCALAPPDATA%")
    progFiles  = shell.ExpandEnvironmentStrings("%ProgramFiles%")
    progFiles86= shell.ExpandEnvironmentStrings("%ProgramFiles(x86)%")

    ' Directories to scan for Python*, pythoncore-*, etc.
    Dim searchRoots, searchRoot
    searchRoots = Array( _
        localApp & "\Programs\Python", _
        localApp & "\Python", _
        progFiles, _
        progFiles86 _
    )

    For Each searchRoot In searchRoots
        If systemPython <> "" Then Exit For
        If fso.FolderExists(searchRoot) Then
            Dim folder, subfolders
            Set subfolders = fso.GetFolder(searchRoot).SubFolders
            ' Collect all matching folders, then pick the newest
            Dim bestFolder, bestName
            bestFolder = ""
            bestName = ""
            For Each folder In subfolders
                Dim fName
                fName = LCase(folder.Name)
                ' Match Python3xx, pythoncore-3.xx-64, etc.
                If Left(fName, 6) = "python" Then
                    Dim candidate
                    candidate = folder.Path & "\python.exe"
                    If fso.FileExists(candidate) Then
                        ' Pick the highest version (latest alphabetically)
                        If fName > bestName Then
                            bestName = fName
                            bestFolder = candidate
                        End If
                    End If
                End If
            Next
            If bestFolder <> "" Then
                systemPython = bestFolder
            End If
        End If
    Next
End If

' 2. Fall back to `py` launcher via PATH
If systemPython = "" Then
    On Error Resume Next
    result = shell.Run("py --version", 0, True)
    On Error GoTo 0
    If result = 0 Then systemPython = "py"
End If


' ═════════════════════════════════════════════
'  RESOLVE pythonw.exe FOR HIDDEN LAUNCH
' ═════════════════════════════════════════════
' Bootstrap now opens its own Tk GUI window, so we launch it
' via pythonw.exe (no console flash) and let it manage its own UI.
Dim launchPython
launchPython = systemPython  ' default fallback

If systemPython <> "" And systemPython <> "py" Then
    Dim pythonwCandidate
    pythonwCandidate = fso.GetParentFolderName(systemPython) & "\pythonw.exe"
    If fso.FileExists(pythonwCandidate) Then
        launchPython = pythonwCandidate
    End If
End If


' ═════════════════════════════════════════════
'  IF NO SYSTEM PYTHON → DOWNLOAD EMBEDDED PYTHON
' ═════════════════════════════════════════════
If systemPython = "" Then
    Dim pythonDir, zipFile, extractFlag
    pythonDir  = scriptDir & "_python"
    zipFile    = scriptDir & "_python_download.zip"
    extractFlag= scriptDir & "_python\.installed"

    ' Show a message so the user knows what's happening
    Dim msgResult
    msgResult = MsgBox( _
        "Python is not installed on this computer." & vbCrLf & vbCrLf & _
        "MCU Uploader IDE by Naph can download a portable Python automatically." & vbCrLf & _
        "(~25 MB download, no admin rights needed)" & vbCrLf & vbCrLf & _
        "Click OK to download, or Cancel to exit.", _
        vbOKCancel + vbInformation, "MCU Uploader IDE by Naph — Setup")

    If msgResult = vbCancel Then
        WScript.Quit 0
    End If

    ' Create _python directory
    If Not fso.FolderExists(pythonDir) Then
        fso.CreateFolder(pythonDir)
    End If

    ' Download Python embeddable zip using PowerShell
    ' Dynamically detects architecture and resolves download URL
    Dim psDownloadCmd

    ' Build PowerShell command that:
    '   1. Detects CPU architecture (amd64 vs arm64)
    '   2. Resolves latest Python 3.12.x embed URL from python.org
    '   3. Downloads and extracts it
    '   4. Enables pip
    psDownloadCmd = "powershell.exe -NoProfile -ExecutionPolicy Bypass -Command """ & _
        "Write-Host ''; " & _
        "Write-Host '  ================================================' -Fore Cyan; " & _
        "Write-Host '   MCU Uploader IDE by Naph - Setting up Python...' -Fore Cyan; " & _
        "Write-Host '  ================================================' -Fore Cyan; " & _
        "Write-Host ''; " & _
        "$arch = if ([Environment]::Is64BitOperatingSystem) { " & _
        "  if ($env:PROCESSOR_ARCHITECTURE -eq 'ARM64') { 'arm64' } else { 'amd64' } " & _
        "} else { 'win32' }; " & _
        "Write-Host '  Architecture:' $arch -Fore Cyan; " & _
        "$pyVer = '3.12.8'; " & _
        "$zipName = ""python-$pyVer-embed-$arch.zip""; " & _
        "$pyUrl = ""https://www.python.org/ftp/python/$pyVer/$zipName""; " & _
        "Write-Host ""  Downloading Python $pyVer ($arch)..."" -Fore Cyan; " & _
        "Write-Host ""  URL: $pyUrl"" -Fore DarkGray; " & _
        "$ProgressPreference = 'Continue'; " & _
        "try { " & _
        "  Invoke-WebRequest -Uri $pyUrl -OutFile '" & zipFile & "' -UseBasicParsing; " & _
        "  Write-Host '  OK Downloaded successfully.' -Fore Green; " & _
        "} catch { " & _
        "  Write-Host '  FAIL Download failed:' $_.Exception.Message -Fore Red; " & _
        "  exit 1; " & _
        "}; " & _
        "Write-Host '  Extracting...' -Fore Cyan; " & _
        "try { " & _
        "  Expand-Archive -Path '" & zipFile & "' -DestinationPath '" & pythonDir & "' -Force; " & _
        "  Write-Host '  OK Extracted to _python folder.' -Fore Green; " & _
        "} catch { " & _
        "  Write-Host '  FAIL Extraction failed:' $_.Exception.Message -Fore Red; " & _
        "  exit 1; " & _
        "}; " & _
        "Remove-Item '" & zipFile & "' -ErrorAction SilentlyContinue; " & _
        "Write-Host ''; " & _
        "Write-Host '  Enabling pip support...' -Fore Cyan; " & _
        "$pthFiles = Get-ChildItem '" & pythonDir & "' -Filter 'python*._pth'; " & _
        "foreach ($f in $pthFiles) { " & _
        "  $content = Get-Content $f.FullName -Raw; " & _
        "  $content = $content -replace '#import site','import site'; " & _
        "  Set-Content $f.FullName $content; " & _
        "  Write-Host '  OK Enabled site-packages in' $f.Name -Fore Green; " & _
        "}; " & _
        "Write-Host ''; " & _
        "Write-Host '  Installing pip...' -Fore Cyan; " & _
        "& '" & pythonDir & "\python.exe' -m ensurepip --upgrade 2>&1 | Out-Null; " & _
        "if ($LASTEXITCODE -ne 0) { " & _
        "  Write-Host '  Trying get-pip.py...' -Fore Yellow; " & _
        "  Invoke-WebRequest -Uri 'https://bootstrap.pypa.io/get-pip.py' -OutFile '" & pythonDir & "\get-pip.py' -UseBasicParsing; " & _
        "  & '" & pythonDir & "\python.exe' '" & pythonDir & "\get-pip.py' 2>&1 | Out-Null; " & _
        "}; " & _
        "Write-Host '  OK pip ready.' -Fore Green; " & _
        "Write-Host ''; " & _
        "Write-Host '  Python setup complete!' -Fore Green; " & _
        "Write-Host ''; " & _
        """"

    Dim dlResult
    dlResult = shell.Run(psDownloadCmd, 1, True)

    If dlResult <> 0 Then
        MsgBox "Python download/setup failed." & vbCrLf & vbCrLf & _
               "Please check your internet connection and try again," & vbCrLf & _
               "or install Python manually from https://www.python.org", _
               vbCritical, "MCU Uploader IDE by Naph"
        WScript.Quit 1
    End If

    ' Verify the download worked
    If fso.FileExists(pythonDir & "\python.exe") Then
        systemPython = pythonDir & "\python.exe"
        ' Portable embeds ship pythonw.exe too — prefer it for the hidden launch
        If fso.FileExists(pythonDir & "\pythonw.exe") Then
            launchPython = pythonDir & "\pythonw.exe"
        Else
            launchPython = systemPython
        End If
    Else
        MsgBox "Python download seemed to succeed but python.exe was not found." & vbCrLf & _
               "Try deleting the _python folder and running again.", _
               vbCritical, "MCU Uploader IDE by Naph"
        WScript.Quit 1
    End If
End If


' ═════════════════════════════════════════════
'  LAUNCH — always via bootstrap
' ═════════════════════════════════════════════
' Bootstrap now runs its own Tk GUI window, so we use pythonw.exe
' (no console flash) and window style 0 (hidden). The bootstrap
' GUI window will appear on its own.
shell.Run """" & launchPython & """ """ & bootstrapFile & """ --hidden", 0, False

' ── Arduino IDE executable names to look for ──
Function IsArduinoExe(fileName)
    Dim lf
    lf = LCase(fileName)
    IsArduinoExe = (lf = "arduino ide.exe") Or (lf = "arduino.exe")
End Function

' ── Check whether a folder directly contains an Arduino exe ──
Function FolderHasArduinoExe(folderPath)
    FolderHasArduinoExe = False
    If Not fso.FolderExists(folderPath) Then Exit Function
    On Error Resume Next
    Dim ff, fe
    Set ff = fso.GetFolder(folderPath)
    For Each fe In ff.Files
        If IsArduinoExe(fe.Name) Then
            FolderHasArduinoExe = True
            Exit Function
        End If
    Next
    On Error GoTo 0
End Function

' ── IsArduinoIdeInstalled ─────────────────────────────────────
' Multi-strategy detection:
'   1. Hard-coded standard locations (ProgramFiles, LOCALAPPDATA …)
'   2. All available drive roots, 2 levels deep
'      (catches D:\Arduino\Arduino IDE\, E:\Tools\Arduino IDE\, etc.)
'   3. Registry uninstall / install-location keys (HKLM + HKCU)
'   4. shutil.which-equivalent via where.exe on PATH
Function IsArduinoIdeInstalled()
    IsArduinoIdeInstalled = False

    Dim sDir
    sDir = fso.GetParentFolderName(WScript.ScriptFullName) & "\"
    Dim cachedPathFile, cachedPath
    cachedPathFile = sDir & "arduino_ide_path.txt"
    If fso.FileExists(cachedPathFile) Then
        On Error Resume Next
        Dim txtStream
        Set txtStream = fso.OpenTextFile(cachedPathFile, 1)
        cachedPath = Trim(txtStream.ReadLine)
        txtStream.Close
        On Error GoTo 0
        If cachedPath <> "" Then
            If fso.FileExists(cachedPath) Then
                IsArduinoIdeInstalled = True
                Exit Function
            End If
        End If
    End If

    Dim pf, pf86, la, up
    pf  = shell.ExpandEnvironmentStrings("%ProgramFiles%")
    pf86= shell.ExpandEnvironmentStrings("%ProgramFiles(x86)%")
    la  = shell.ExpandEnvironmentStrings("%LOCALAPPDATA%")
    up  = shell.ExpandEnvironmentStrings("%USERPROFILE%")

    ' ── Strategy 1: standard / well-known paths ────────────────
    Dim stdPaths, p
    stdPaths = Array( _
        pf    & "\Arduino IDE\Arduino IDE.exe", _
        pf    & "\Arduino\Arduino IDE.exe", _
        pf    & "\Arduino\arduino.exe", _
        pf86  & "\Arduino IDE\Arduino IDE.exe", _
        pf86  & "\Arduino\arduino.exe", _
        la    & "\Programs\Arduino IDE\Arduino IDE.exe", _
        la    & "\Programs\Arduino\arduino.exe", _
        la    & "\Arduino IDE\Arduino IDE.exe", _
        up    & "\AppData\Local\Programs\Arduino IDE\Arduino IDE.exe", _
        up    & "\Arduino IDE\Arduino IDE.exe", _
        up    & "\Arduino\arduino.exe" _
    )

    For Each p In stdPaths
        If fso.FileExists(p) Then
            IsArduinoIdeInstalled = True
            Exit Function
        End If
    Next

    ' ── Strategy 2: scan all drive roots, 2 levels deep ────────
    ' Uses FSO.Drives to enumerate all present drives automatically.
    Dim drv, topFolder, subFolder, subFolders, topFolders
    For Each drv In fso.Drives
        ' Only scan drives that are ready and fixed/removable (not CD/network)
        On Error Resume Next
        Dim drvReady
        drvReady = drv.IsReady
        On Error GoTo 0
        If drvReady Then
            On Error Resume Next
            Set topFolders = fso.GetFolder(drv.RootFolder).SubFolders
            On Error GoTo 0
            If Not IsEmpty(topFolders) Then
                For Each topFolder In topFolders
                    ' Level 1: drive\FolderName\Arduino IDE.exe
                    If FolderHasArduinoExe(topFolder.Path) Then
                        IsArduinoIdeInstalled = True
                        Exit Function
                    End If
                    ' Level 2: drive\FolderName\SubFolder\Arduino IDE.exe
                    On Error Resume Next
                    Set subFolders = topFolder.SubFolders
                    On Error GoTo 0
                    If Not IsEmpty(subFolders) Then
                        For Each subFolder In subFolders
                            If FolderHasArduinoExe(subFolder.Path) Then
                                IsArduinoIdeInstalled = True
                                Exit Function
                            End If
                        Next
                    End If
                Next
            End If
        End If
    Next

    ' ── Strategy 3: Registry uninstall / install-location keys ─
    Dim regKeys, k, val
    regKeys = Array( _
        "HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\Arduino IDE\InstallLocation", _
        "HKLM\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\Arduino IDE\InstallLocation", _
        "HKCU\Software\Microsoft\Windows\CurrentVersion\Uninstall\Arduino IDE\InstallLocation", _
        "HKLM\SOFTWARE\WOW6432Node\Arduino\Install_Dir", _
        "HKLM\SOFTWARE\Arduino\Install_Dir" _
    )

    On Error Resume Next
    For Each k In regKeys
        val = shell.RegRead(k)
        If Err.Number = 0 And val <> "" Then
            ' InstallLocation may point to the parent or the subfolder
            If FolderHasArduinoExe(val) Then
                IsArduinoIdeInstalled = True
                On Error GoTo 0
                Exit Function
            End If
            ' One level down
            If fso.FolderExists(val) Then
                Dim sf2
                On Error Resume Next
                Dim subFolders2
                Set subFolders2 = fso.GetFolder(val).SubFolders
                For Each sf2 In subFolders2
                    If FolderHasArduinoExe(sf2.Path) Then
                        IsArduinoIdeInstalled = True
                        On Error GoTo 0
                        Exit Function
                    End If
                Next
                On Error GoTo 0
            End If
        End If
        Err.Clear
    Next
    On Error GoTo 0



    ' ── Strategy 4: PATH lookup via where.exe ───────────────────
    Dim whereResult
    On Error Resume Next
    whereResult = shell.Run("where arduino-ide.exe", 0, True)
    If whereResult = 0 Then
        IsArduinoIdeInstalled = True
        On Error GoTo 0
        Exit Function
    End If
    whereResult = shell.Run("where arduino.exe", 0, True)
    If whereResult = 0 Then
        IsArduinoIdeInstalled = True
        On Error GoTo 0
        Exit Function
    End If
    On Error GoTo 0
End Function


' ── Helper to manually choose Arduino IDE executable ──────────
Function ChooseArduinoExe()
    Dim tempFile, psCmd
    tempFile = shell.ExpandEnvironmentStrings("%TEMP%\arduino_path_sel.txt")
    
    psCmd = "powershell -NoProfile -WindowStyle Hidden -Command ""Add-Type -AssemblyName System.Windows.Forms; " & _
            "$f = New-Object System.Windows.Forms.OpenFileDialog; " & _
            "$f.Filter = 'Arduino Executable (Arduino IDE.exe;arduino.exe)|Arduino IDE.exe;arduino.exe'; " & _
            "$f.Title = 'Select Arduino IDE Executable'; " & _
            "$res = if ($f.ShowDialog() -eq 'OK') { $f.FileName } else { '' }; " & _
            "$res | Out-File -FilePath '" & tempFile & "' -Encoding ascii"""
            
    shell.Run psCmd, 0, True
    
    ChooseArduinoExe = ""
    If fso.FileExists(tempFile) Then
        Dim txtStream
        Set txtStream = fso.OpenTextFile(tempFile, 1)
        On Error Resume Next
        ChooseArduinoExe = Trim(txtStream.ReadLine)
        txtStream.Close
        fso.DeleteFile tempFile
        On Error GoTo 0
    End If
End Function


' ── Helper to run offline installer MSI ────────────────────────
Sub InstallArduinoMsi(scriptDir)
    Dim msiPath
    msiPath = scriptDir & "installers\arduino-ide.msi"
    If Not fso.FileExists(msiPath) Then
        MsgBox "Arduino IDE installer MSI not found at:" & vbCrLf & msiPath, vbExclamation, "Installer Not Found"
        Exit Sub
    End If
    
    MsgBox "Launching Arduino IDE installer. Please complete the installation wizard.", vbInformation, "Arduino IDE Installer"
    shell.Run "msiexec /i """ & msiPath & """", 1, True
End Sub
